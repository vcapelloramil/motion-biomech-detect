# mediapipe 0.10.x exige numpy<2 -> numpy queda en la serie 1.26.

# mediapipe depende de opencv-contrib-python (no de opencv-python). contrib es
# superset de opencv-python, así que se usa ese y no los dos a la vez: tener
# ambos instalados pisa los archivos de cv2 y deja el import roto.
opencv-contrib-python==4.11.0.86
mediapipe==0.10.18

fastapi==0.115.5
uvicorn==0.32.1
# 2.10.3 -> 2.11.7 al agregar supabase==2.31.0 (Etapa 4.5): su dependencia `realtime`
# exige pydantic>=2.11.7. Es el piso mínimo que resuelve el conflicto, no una
# actualización oportunista a la última (2.13.x); fastapi y el contrato de
# backend/app/schemas/reporte.py siguen funcionando igual (es un salto menor dentro de
# Pydantic v2, no v1->v2).
pydantic==2.11.7
python-dotenv==1.0.1

supabase==2.31.0
